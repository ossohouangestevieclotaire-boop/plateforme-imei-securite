async function forgotPassword() { const email = prompt("Entrez votre adresse e-mail enregistrée :"); if (!email) return; try { const response = await fetch('/api/index', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'forgot-password', email }) }); const data = await response.json(); if (response.ok) { alert(data.message); } else { alert("Erreur : " + (data.error || "Impossible d'envoyer le lien.")); } } catch (err) { alert("Erreur de connexion avec le serveur."); } }
